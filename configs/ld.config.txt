# Linker namespace configuration for LG H872 MSM8996
# Allows vendor processes (camera daemon) to access system libraries like libandroid.so
# 
# Without this configuration, the linker blocks vendor executables from seeing
# system libraries, causing camera daemon to fail with:
# "CANNOT LINK EXECUTABLE ... library libandroid.so not found"

# Default namespace - system libraries
namespace.default {
  search.paths = /system/lib:/system/lib64:/vendor/lib:/vendor/lib64:/system/vendor/lib:/system/vendor/lib64:/apex/com.android.i18n/lib:/apex/com.android.i18n/lib64:/apex/com.android.i18n@1/lib:/apex/com.android.i18n@1/lib64
  permitted.paths = /system/lib:/system/lib64:/vendor/lib:/vendor/lib64:/system/vendor/lib:/system/vendor/lib64:/apex/com.android.i18n/lib:/apex/com.android.i18n/lib64:/apex/com.android.i18n@1/lib:/apex/com.android.i18n@1/lib64
}

namespace.vendor {
  search.paths = /vendor/lib:/vendor/lib64:/system/vendor/lib:/system/vendor/lib64:/system/lib:/system/lib64:/apex/com.android.i18n/lib:/apex/com.android.i18n/lib64:/apex/com.android.i18n@1/lib:/apex/com.android.i18n@1/lib64
  permitted.paths = /vendor/lib:/vendor/lib64:/system/vendor/lib:/system/vendor/lib64:/system/lib:/system/lib64:/apex/com.android.i18n/lib:/apex/com.android.i18n/lib64:/apex/com.android.i18n@1/lib:/apex/com.android.i18n@1/lib64
  links = default
  link.default.shared_libs = libandroid.so:libandroidicu.so:libxml2.so:libicui18n.so:libicuuc.so:libc.so:libm.so:libdl.so:liblog.so
}
