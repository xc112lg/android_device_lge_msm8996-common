# Linker namespace configuration for LG H872 MSM8996
# Allows vendor processes (camera daemon) to access system libraries like libandroid.so
# 
# Without this configuration, the linker blocks vendor executables from seeing
# system libraries, causing camera daemon to fail with:
# "CANNOT LINK EXECUTABLE ... library libandroid.so not found"

# Default namespace - system libraries
namespace.default {
  search.paths = /system/lib:/system/lib64:/vendor/lib:/vendor/lib64
  permitted.paths = /system/lib:/system/lib64:/vendor/lib:/vendor/lib64:/system/vendor/lib:/system/vendor/lib64
}

# Vendor namespace - for /vendor/bin executables like camera daemon
namespace.vendor {
  search.paths = /vendor/lib:/vendor/lib64:/system/vendor/lib:/system/vendor/lib64:/system/lib:/system/lib64
  permitted.paths = /vendor/lib:/vendor/lib64:/system/vendor/lib:/system/vendor/lib64:/system/lib:/system/lib64
  links = default
  # Allow vendor to link these critical system libraries
  link.default.shared_libs = libandroid.so:libcutils.so:libdl.so:liblog.so:libc.so:libm.so:libstdc++.so:libgui.so:libui.so
}
